cmake_minimum_required(VERSION 3.20)
project(VeloSim LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(VELOSIM_BUILD_UHDM "Build tools against an installed matching UHDM" OFF)
option(VELOSIM_COVERAGE "Instrument VeloSim C++ with LLVM source coverage" OFF)
option(VELOSIM_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(velosim_options INTERFACE)
if(VELOSIM_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "LLVM coverage requires Clang")
  endif()
  target_compile_options(velosim_options INTERFACE -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(velosim_options INTERFACE -fprofile-instr-generate)
endif()
if(VELOSIM_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Sanitizer configuration currently requires GCC or Clang")
  endif()
  target_compile_options(velosim_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(velosim_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(velosim_ir SHARED native/ir/engine.cpp)
add_library(velosim_scheduler SHARED native/scheduler.cpp)
add_library(velosim_vpi SHARED native/rtl/vpi.cpp)
target_compile_definitions(velosim_vpi PRIVATE VELOSIM_VPI_BUILD)
target_include_directories(velosim_vpi PUBLIC native/rtl)
target_link_libraries(velosim_vpi PRIVATE velosim_options ${CMAKE_DL_LIBS})
foreach(target velosim_ir velosim_scheduler)
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  target_include_directories(${target} PUBLIC native/include)
  target_link_libraries(${target} PRIVATE Threads::Threads velosim_options)
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(velosim_ir PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/native/ir/exports.map")
  target_link_options(velosim_scheduler PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/native/scheduler.exports.map")
  set_property(TARGET velosim_ir APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/ir/exports.map")
  set_property(TARGET velosim_scheduler APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/scheduler.exports.map")
endif()
include(CTest)
if(BUILD_TESTING)
  add_library(velosim_external_plugin MODULE examples/external_plugin.c)
  target_compile_definitions(velosim_external_plugin PRIVATE VELOSIM_PLUGIN_TOP="test")
  target_link_libraries(velosim_external_plugin PRIVATE velosim_vpi velosim_options)
  add_executable(velosim_external_tests tests/native/external_tests.cpp)
  target_link_libraries(velosim_external_tests PRIVATE velosim_vpi velosim_options Threads::Threads)
  add_test(NAME external_interfaces COMMAND velosim_external_tests $<TARGET_FILE:velosim_external_plugin>)
  add_executable(velosim_rtl_tests tests/native/rtl_runtime_tests.cpp)
  target_link_libraries(velosim_rtl_tests PRIVATE velosim_options)
  add_test(NAME rtl_runtime COMMAND velosim_rtl_tests)
  add_executable(velosim_native_tests tests/native/abi_tests.cpp)
  target_link_libraries(velosim_native_tests PRIVATE velosim_ir velosim_scheduler velosim_options)
  add_test(NAME native_abi COMMAND velosim_native_tests)
  add_executable(velosim_c_abi_tests tests/native/c_abi_tests.c)
  set_target_properties(velosim_c_abi_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
  target_link_libraries(velosim_c_abi_tests PRIVATE velosim_ir velosim_scheduler velosim_options)
  add_test(NAME c_abi COMMAND velosim_c_abi_tests)
  add_executable(velosim_word_tests tests/native/word_tests.cpp)
  target_link_libraries(velosim_word_tests PRIVATE velosim_ir velosim_options)
  add_test(NAME word_abi COMMAND velosim_word_tests)
  add_executable(velosim_json_tests tests/test_json_escape.cpp)
  target_link_libraries(velosim_json_tests PRIVATE velosim_options)
  if(MSVC)
    target_compile_options(velosim_json_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(velosim_json_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME json_escape COMMAND velosim_json_tests)
endif()
if(VELOSIM_BUILD_UHDM)
  add_subdirectory(native/uhdm)
endif()
install(TARGETS velosim_ir velosim_scheduler velosim_vpi LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES native/include/velosim.h DESTINATION include)
install(FILES native/rtl/runtime.h native/rtl/external_model.h native/rtl/vpi_user.h DESTINATION include/velosim)
