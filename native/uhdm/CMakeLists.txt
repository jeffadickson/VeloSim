cmake_minimum_required(VERSION 3.20)
project(velosim_uhdm_inspect LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(UHDM CONFIG QUIET)
if(TARGET uhdm::uhdm)
  set(VELOSIM_UHDM_TARGET uhdm::uhdm)
else()
  find_path(UHDM_INCLUDE_DIR uhdm/uhdm.h REQUIRED)
  find_library(UHDM_LIBRARY NAMES uhdm UHDM REQUIRED)
  add_library(velosim_uhdm_dependency UNKNOWN IMPORTED)
  set_target_properties(velosim_uhdm_dependency PROPERTIES
    IMPORTED_LOCATION "${UHDM_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${UHDM_INCLUDE_DIR}")
  set(VELOSIM_UHDM_TARGET velosim_uhdm_dependency)
endif()
add_executable(velosim-uhdm-inspect inspect.cpp)
target_link_libraries(velosim-uhdm-inspect PRIVATE ${VELOSIM_UHDM_TARGET})

add_executable(velosim-uhdm-normalize normalize.cpp)
target_link_libraries(velosim-uhdm-normalize PRIVATE ${VELOSIM_UHDM_TARGET})
add_executable(velosim-uhdm-rtl rtl.cpp)
target_link_libraries(velosim-uhdm-rtl PRIVATE ${VELOSIM_UHDM_TARGET})
option(VELOSIM_UHDM_PCH "Cache installed UHDM headers for repeated exporter builds" OFF)
if(VELOSIM_UHDM_PCH)
  # Private build acceleration only; consumers retain their own header policy.
  target_precompile_headers(velosim-uhdm-rtl PRIVATE <uhdm/uhdm.h>)
endif()
if(TARGET velosim_options)
  target_link_libraries(velosim-uhdm-inspect PRIVATE velosim_options)
  target_link_libraries(velosim-uhdm-normalize PRIVATE velosim_options)
  target_link_libraries(velosim-uhdm-rtl PRIVATE velosim_options)
  if(VELOSIM_COVERAGE)
    target_compile_definitions(velosim-uhdm-rtl PRIVATE VELOSIM_EXPORTER_COVERAGE)
  endif()
endif()

install(TARGETS velosim-uhdm-inspect velosim-uhdm-normalize velosim-uhdm-rtl RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  add_executable(velosim_uhdm_audit_tests ../../tests/native/uhdm_audit_tests.cpp)
  target_link_libraries(velosim_uhdm_audit_tests PRIVATE ${VELOSIM_UHDM_TARGET} velosim_options)
  add_test(NAME uhdm_executable_audit COMMAND velosim_uhdm_audit_tests)
  add_executable(velosim_uhdm_type_tests ../../tests/native/uhdm_type_budget_tests.cpp)
  target_link_libraries(velosim_uhdm_type_tests PRIVATE ${VELOSIM_UHDM_TARGET} velosim_options)
  add_test(NAME uhdm_type_budget COMMAND velosim_uhdm_type_tests)
  # UHDM itself is uninstrumented; this test checks bounded work and ASan/UBSan.
  set_tests_properties(uhdm_type_budget PROPERTIES TIMEOUT 10 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1")
endif()
