cmake_minimum_required(VERSION 3.20)
project(VeloSimFuzz LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "This optional campaign requires Clang/libFuzzer")
endif()
foreach(domain RANGE 0 3)
  add_executable(fuzz_${domain} runtime_fuzz.cpp)
  target_compile_definitions(fuzz_${domain} PRIVATE FUZZ_DOMAIN=${domain})
  target_compile_options(fuzz_${domain} PRIVATE -g -O1 -fno-omit-frame-pointer -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz_${domain} PRIVATE -fsanitize=fuzzer,address,undefined)
  add_executable(replay_${domain} runtime_fuzz.cpp)
  target_compile_definitions(replay_${domain} PRIVATE FUZZ_DOMAIN=${domain} VELOSIM_FUZZ_REPLAY)
  target_compile_options(replay_${domain} PRIVATE -g -O0 -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(replay_${domain} PRIVATE -fprofile-instr-generate)
endforeach()
