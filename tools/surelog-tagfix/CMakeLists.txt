cmake_minimum_required(VERSION 3.20)
project(velosim_surelog_tagfix LANGUAGES CXX)
# Compile only the two corrected translation units against the pinned native SDK.
# Input hashes make this overlay fail closed when upstream sources change.
set(SURELOG_SOURCE "" CACHE PATH "Pinned Surelog source checkout")
find_package(Threads REQUIRED)
find_package(Surelog CONFIG REQUIRED PATHS /opt/velosim/lib/cmake/Surelog NO_DEFAULT_PATH)
if(NOT Surelog_DIR STREQUAL "/opt/velosim/lib/cmake/Surelog")
  message(FATAL_ERROR "Overlay requires the matching /opt/velosim SDK")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SURELOG_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL "6db44ea2048c8ab9c702e389c2897f7e2318ccc5")
  message(FATAL_ERROR "Overlay requires the pinned Surelog revision")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SURELOG_SOURCE}" diff --exit-code HEAD -- include src RESULT_VARIABLE dirty OUTPUT_QUIET)
if(NOT dirty EQUAL 0)
  message(FATAL_ERROR "Overlay requires clean upstream source and headers")
endif()
# Record the complete effective native SDK inputs, including headers not installed
# by Surelog itself. This manifest accompanies the generated source and executable.
file(GLOB_RECURSE sdk_inputs LIST_DIRECTORIES false "/opt/velosim/include/*" "/opt/velosim/lib/*.a" "${SURELOG_SOURCE}/include/*")
list(SORT sdk_inputs)
set(manifest "revision ${revision}\ncompiler ${CMAKE_CXX_COMPILER}\n")
foreach(input IN LISTS sdk_inputs)
  file(SHA256 "${input}" sha)
  string(APPEND manifest "${sha}  ${input}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-inputs.sha256" "${manifest}")
function(pinned_copy relative expected output)
  file(SHA256 "${SURELOG_SOURCE}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Pinned source hash mismatch: ${relative}")
  endif()
  file(READ "${SURELOG_SOURCE}/${relative}" content)
  set(${output} "${content}" PARENT_SCOPE)
endfunction()
pinned_copy(src/DesignCompile/CompileType.cpp 82f9b97fef429160b629ced29dcf23fddd61d60394feff290413691bc218060c type_source)
pinned_copy(src/DesignCompile/UhdmWriter.cpp 8e58ecfcd5ebe9d7eb46d0d7be0e1408472b6c64baac2892d5b970603fa50dda writer_source)
pinned_copy(src/main.cpp d90f264446146c063f7d2ea8c35de73ce98e30ac47ab5b372a5ad21174eed698 main_source)
string(REPLACE "union_typespec* ts = s.MakeUnion_typespec();" "union_typespec* ts = s.MakeUnion_typespec();\n        // Preserve tagged syntax that the pinned frontend otherwise discards.\n        ts->VpiTagged(fC->Type(fC->Sibling(struct_or_union)) == VObjectType::paTagged_keyword);" type_source "${type_source}")
string(REPLACE "s.Save(uhdmFile);" "// Persist a capability marker only after elaboration, outside HDL namespaces.\n    if (d) {\n      auto marker = s.MakeVoid_typespec();\n      marker->VpiName(\"velosim frontend capability: union tags v1\");\n      marker->VpiParent(d);\n      if (!d->Typespecs()) d->Typespecs(s.MakeTypespecVec());\n      d->Typespecs()->push_back(marker);\n    }\n    s.Save(uhdmFile);" writer_source "${writer_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CompileType.cpp" "${type_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UhdmWriter.cpp" "${writer_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" "${main_source}")
add_executable(velosim-surelog "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/CompileType.cpp" "${CMAKE_CURRENT_BINARY_DIR}/UhdmWriter.cpp")
target_include_directories(velosim-surelog PRIVATE "${SURELOG_SOURCE}/include")
target_compile_features(velosim-surelog PRIVATE cxx_std_17)
target_link_libraries(velosim-surelog PRIVATE surelog::surelog)
# Keep the conventional name inside this private directory.
set_target_properties(velosim-surelog PROPERTIES OUTPUT_NAME surelog)
