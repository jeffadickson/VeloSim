# Record the complete effective native SDK inputs, including headers not installed
# by Surelog itself. This manifest accompanies the generated source and executable.
file(GLOB_RECURSE sdk_inputs LIST_DIRECTORIES false "/opt/velosim/include/*" "/opt/velosim/lib/*.a" "${SURELOG_SOURCE}/include/*")
list(SORT sdk_inputs)
set(manifest "revision ${revision}\ncompiler ${CMAKE_CXX_COMPILER}\n")
foreach(input IN LISTS sdk_inputs)
  file(SHA256 "${input}" sha)
  string(APPEND manifest "${sha}  ${input}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-inputs.sha256" "${manifest}")
function(pinned_copy relative expected output)
  file(SHA256 "${SURELOG_SOURCE}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Pinned source hash mismatch: ${relative}")
  endif()
  file(READ "${SURELOG_SOURCE}/${relative}" content)
  set(${output} "${content}" PARENT_SCOPE)
endfunction()
# Preserve timestamps for unchanged generated translation units. Reconfiguring
# an overlay must not force all large upstream units to compile again.
function(write_generated path content)
  if(EXISTS "${path}")
    file(READ "${path}" previous)
    if(previous STREQUAL content)
      return()
    endif()
  endif()
  file(WRITE "${path}" "${content}")
endfunction()
